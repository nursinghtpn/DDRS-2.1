// AUTO SETUP DATABASE + AUTO MIGRATE
// ============================================================
function autoSetupDatabase() {
  const ss     = SpreadsheetApp.getActiveSpreadsheet();
  const log_   = [];

  const sheets = [
    { name: SHEET_NAME_REKOD,    headers: HEADERS_REKOD,    bg: '#1e3a5f', fg: '#ffffff', empty: '-' },
    { name: SHEET_NAME_PENGGUNA, headers: HEADERS_PENGGUNA, bg: '#2d6a4f', fg: '#ffffff', empty: '' },
    { name: SHEET_NAME_LOG,      headers: HEADERS_LOG,      bg: '#7b2d8b', fg: '#ffffff', empty: '' },
    { name: SHEET_NAME_CONFIG,   headers: HEADERS_CONFIG,   bg: '#555555', fg: '#ffffff', empty: '' },
    { name: SHEET_NAME_JENIS,    headers: ['Nama_Jenis','Status'], bg: '#0369a1', fg: '#ffffff', empty: '' },
  ];

  sheets.forEach(cfg => {
    const result = _syncSheet(ss, cfg.name, cfg.headers, cfg.bg, cfg.fg, cfg.empty);
    log_.push(result);
  });

  _seedDefaultUsers(ss);
  _seedConfig(ss);

  return log_.join('\n');
}

// _syncSheet — cipta sheet baharu ATAU kemaskini sheet sedia ada
// Strategi: semak nama header mengikut posisi DAN senarai nama yang ada
function _syncSheet(ss, name, headers, headerBg, headerFg, emptyVal) {
  let sheet = ss.getSheetByName(name);
  let isNew  = false;

  // ── 1. CIPTA SHEET BARU jika tiada ──────────────────────
  if (!sheet) {
    sheet = ss.insertSheet(name);
    const hRange = sheet.getRange(1, 1, 1, headers.length);
    hRange.setValues([headers]);
    hRange.setBackground(headerBg);
    hRange.setFontColor(headerFg);
    hRange.setFontWeight('bold');
    hRange.setFontSize(10);
    sheet.setFrozenRows(1);
    sheet.setColumnWidths(1, headers.length, 160);
    Logger.log('[' + name + '] (BARU) - ' + headers.length + ' kolum dicipta');
    return name + ' (BARU)';
  }

  // ── 2. BACA HEADER SEDIA ADA ────────────────────────────
  const lastCol       = Math.max(sheet.getLastColumn(), 1);
  const existingRow   = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  const existingNames = existingRow.map(h => String(h).trim()); // senarai nama sedia ada
  const changes       = [];

  headers.forEach((headerName, idx) => {
    const col     = idx + 1;
    const current = existingNames[idx] || '';

    // Nama sudah betul pada posisi ini — skip
    if (current === headerName) return;

    // Nama header ini sudah ada pada posisi lain — jangan overwrite (data mungkin ada)
    if (existingNames.includes(headerName)) return;

    // Kolum kosong atau belum ada — tambah header baharu
    const cell = sheet.getRange(1, col);
    cell.setValue(headerName);
    cell.setBackground(headerBg);
    cell.setFontColor(headerFg);
    cell.setFontWeight('bold');
    cell.setFontSize(10);
    sheet.setColumnWidth(col, 160);

    // Isi nilai lalai untuk data sedia ada
    const lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.getRange(2, col, lastRow - 1, 1).setValue(emptyVal !== undefined ? emptyVal : '-');
    }

    changes.push('+ ' + headerName + ' (kol ' + col + ')');
  });

  // ── 3. TAMBAH KOLUM HUJUNG jika jumlah header bertambah ─
  const existingFilled = existingNames.filter(h => h !== '').length;
  for (let i = existingFilled; i < headers.length; i++) {
    const headerName = headers[i];
    if (existingNames.includes(headerName)) continue;

    const col  = i + 1;
    const cell = sheet.getRange(1, col);
    cell.setValue(headerName);
    cell.setBackground(headerBg);
    cell.setFontColor(headerFg);
    cell.setFontWeight('bold');
    cell.setFontSize(10);
    sheet.setColumnWidth(col, 160);

    const lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.getRange(2, col, lastRow - 1, 1).setValue(emptyVal !== undefined ? emptyVal : '-');
    }

    if (!changes.some(c => c.includes(headerName))) {
      changes.push('+ ' + headerName + ' (kol ' + col + ')');
    }
  }

  const status = changes.length > 0
    ? '(DIKEMASKINI: ' + changes.length + ' kolum baru)'
    : '(tiada perubahan)';

  if (changes.length > 0) Logger.log('[' + name + '] ' + changes.join(', '));
  return name + ' ' + status;
}

function doLogin(emel, password) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const data  = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    // Login menggunakan emel (kolum 6, index 5) dan kata laluan (kolum 3, index 2)
    if (String(row[5]).trim().toLowerCase() === String(emel).trim().toLowerCase() &&
        String(row[2]).trim() === String(password).trim()) {

      // Kolum 10 (index 9) = Status_Akaun. Kosong/tiada dianggap Aktif (akaun lama/admin)
      const status = String(row[9] || STATUS_AKAUN.AKTIF).trim() || STATUS_AKAUN.AKTIF;

      if (status === STATUS_AKAUN.MENUNGGU) {
        return { pending: true, status: 'menunggu', message: 'Akaun anda masih menunggu kelulusan Pentadbir Sistem.' };
      }
      if (status === STATUS_AKAUN.DITOLAK) {
        return { pending: true, status: 'ditolak', message: 'Permohonan akaun anda telah ditolak. Sila hubungi Pentadbir Sistem.' };
      }
      return {
        id:      row[0],
        name:    row[1],
        role:    row[3],
        modules: String(row[4]).split(',').map(m => m.trim()),
        unit:    row[10] || ''
      };
    }
  }
  return null;
}

// Permohonan Akaun Baharu — pendaftaran kendiri, status awal "Menunggu Kelulusan"
function daftarAkaunBaru(data) {
  autoSetupDatabase();
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();

  const emelBersih = String(data.emel || '').trim().toLowerCase();
  if (!emelBersih) return { success:false, message:'Emel tidak sah.' };
  if (!data.password || String(data.password).length < 6) {
    return { success:false, message:'Kata laluan mesti sekurang-kurangnya 6 aksara.' };
  }

  // Semak emel sudah wujud
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][5] || '').trim().toLowerCase() === emelBersih) {
      return { success:false, message:'Emel ini sudah didaftarkan. Sila log masuk atau hubungi Pentadbir.' };
    }
  }

  const id  = 'user' + Date.now().toString().slice(-6) + Math.floor(Math.random()*900+100);
  const now = new Date().toISOString();
  sheet.appendRow([
    id, data.nama, data.password, 'user', 'hantar,status',
    data.emel, data.tel || '', data.jawatan || '', now, STATUS_AKAUN.MENUNGGU, data.unit || ''
  ]);

  return { success:true, id };
}

// Dapatkan senarai akaun yang menunggu kelulusan (untuk Pentadbir)
function dapatkanPermohonanMenunggu() {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const data  = sheet.getDataRange().getValues();
  return data.slice(1).filter(r => r[0] !== '' && String(r[9]).trim() === STATUS_AKAUN.MENUNGGU).map(r => ({
    id: r[0], name: r[1], role: r[3], emel: r[5], tel: r[6], jawatan: r[7], tarikhKemaskini: r[8]
  }));
}

// Pentadbir meluluskan / menolak permohonan akaun
function luluskanAkaun(id, tindakan) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();
  const statusBaru = tindakan === 'LULUS' ? STATUS_AKAUN.AKTIF : STATUS_AKAUN.DITOLAK;
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(id).trim()) {
      sheet.getRange(i+1, 10).setValue(statusBaru);
      sheet.getRange(i+1, 9).setValue(new Date().toISOString());
      break;
    }
  }
  return dapatkanPermohonanMenunggu();
}

  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_REKOD);
  
  const row = [
    data.id,                   // ID_Dokumen      (1)
    data.jenisDokumen || '-',  // Jenis_Dokumen   (2)
    data.namaDokumen,          // Nama_Dokumen     (3)
    data.namaPenghantar,       // Nama_Penghantar  (4)
    data.idPenghantar,         // ID_Penghantar    (5)
    data.tarikhSerahan,        // Tarikh_Hantar    (6)
    STATUS.DIHANTAR,           // Status           (7)
    data.catatan || '-',       // Catatan_Penghantar (8)
    '-', '-', '-', '-',        // Penyemak: Nama,Tarikh,Tindakan,Catatan (9-12)
    '-', '-',                  // Penerima USM: ID, Nama (13-14)
    '-', '-', '-', '-',        // USM: Nama,Tarikh,Tindakan,Catatan (15-18)
    new Date().toISOString(),  // Tarikh_Kemaskini (19)
    data.unitPenghantar || '-' // Unit_Penghantar  (20)
  ];
  
  sheet.appendRow(row);
  _catatLog(ss, data.id, data.namaDokumen, data.namaPenghantar, 'user', 
            'HANTAR', `Dokumen dihantar untuk semakan. Catatan: ${data.catatan || '-'}`);
  
  return dapatkanSenarai();
}

// Unit Kejururawatan: Terima (→ Disemak) atau Tolak (→ Ditolak)
function tindakanPenyemak(docId, tindakan, namaPenyemak, catatanPenyemak) {
  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_REKOD);
  const data = sheet.getDataRange().getValues();
  
  const tarikhSekarang = new Date().toLocaleString('ms-MY', {
    day: '2-digit', month: 'short', year: 'numeric', 
    hour: '2-digit', minute: '2-digit'

// Unit Kejururawatan / USM: Kembalikan sebagai Query kepada penghantar
function tolakQuery(docId, jenis, namaPenolak, alasan, arahan) {
  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_REKOD);
  const data = sheet.getDataRange().getValues();

  const tarikhSekarang = new Date().toLocaleString('ms-MY', {
    day: '2-digit', month: 'short', year: 'numeric',
    hour: '2-digit', minute: '2-digit'
  });
  const catatanGabung = alasan + (arahan ? ' | Arahan: ' + arahan : '');
  const statusBaru = jenis === 'usm' ? STATUS.QUERY_USM : STATUS.QUERY;
  const tindakanLog = jenis === 'usm' ? 'USM_QUERY' : 'PENYEMAK_QUERY';

  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === docId) {
      sheet.getRange(i + 1, 7).setValue(statusBaru);             // Status
      if (jenis === 'usm') {
        sheet.getRange(i + 1, 15).setValue(namaPenolak);         // Nama_USM
        sheet.getRange(i + 1, 16).setValue(tarikhSekarang);      // Tarikh_USM
        sheet.getRange(i + 1, 17).setValue('QUERY');             // Tindakan_USM
        sheet.getRange(i + 1, 18).setValue(catatanGabung);       // Catatan_USM
      } else {
        sheet.getRange(i + 1, 9).setValue(namaPenolak);          // Nama_Penyemak
        sheet.getRange(i + 1, 10).setValue(tarikhSekarang);      // Tarikh_Semak
        sheet.getRange(i + 1, 11).setValue('QUERY');             // Tindakan_Penyemak
        sheet.getRange(i + 1, 12).setValue(catatanGabung);       // Catatan_Penyemak
      }
      sheet.getRange(i + 1, 19).setValue(new Date().toISOString()); // Tarikh_Kemaskini
      _catatLog(ss, docId, data[i][2], namaPenolak, jenis === 'usm' ? 'usm' : 'penyemak',
                tindakanLog, `Sebab: ${alasan}${arahan ? ' | Arahan: ' + arahan : ''}`);
      break;
    }
  }
  return dapatkanSenarai();
}

function kemaskiniProfil(data) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(data.id).trim()) {
      const emelSediaAda = rows[i][5] || '';
      const emelBaharu   = (data.emel !== null && data.emel !== undefined) ? data.emel : emelSediaAda;
      sheet.getRange(i+1, 2).setValue(data.name    || rows[i][1]);
      sheet.getRange(i+1, 6).setValue(emelBaharu);
      sheet.getRange(i+1, 7).setValue(data.tel     || '');
      sheet.getRange(i+1, 8).setValue(data.jawatan || '');
      sheet.getRange(i+1, 9).setValue(new Date().toISOString());
      sheet.getRange(i+1, 11).setValue(data.unit   || '');  // Unit (kolum 11)
      return {
        success: true,
        user: { id:data.id, name:data.name, role:rows[i][3],
                modules: String(rows[i][4]).split(',').map(m=>m.trim()),
                emel: emelBaharu, tel:data.tel, jawatan:data.jawatan, unit:data.unit }
      };
    }
  }
  return { success: false, message: 'Pengguna tidak dijumpai.' };
}

function tukarKataLaluan(userId, passLama, passBaharu) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]).trim() === String(userId).trim()) {
      if (String(rows[i][2]).trim() !== String(passLama).trim()) {
        return { success: false, message: 'Kata laluan semasa tidak betul.' };
      }
      sheet.getRange(i+1, 3).setValue(passBaharu);
      sheet.getRange(i+1, 9).setValue(new Date().toISOString());
      return { success: true };
    }
  }
  return { success: false, message: 'Pengguna tidak dijumpai.' };
}

// Seed sheet Jenis_Dokumen dengan senarai default
function _seedJenisDokumen(ss) {
  const sheet = ss.getSheetByName(SHEET_NAME_JENIS);
  if (!sheet || sheet.getLastRow() > 1) return; // Sudah ada data — jangan overwrite
  const rows = SENARAI_JENIS_DEFAULT.map(nama => [nama, 'Aktif']);
  sheet.getRange(2, 1, rows.length, 2).setValues(rows);
}

// Ambil semua jenis dokumen yang Aktif dari sheet Jenis_Dokumen
function dapatkanJenis() {
  autoSetupDatabase();
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_JENIS);
  if (!sheet || sheet.getLastRow() <= 1) {
    _seedJenisDokumen(ss);
    return SENARAI_JENIS_DEFAULT;
  }
  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).getValues();
  return data
    .filter(r => r[0] !== '' && String(r[1]).trim().toLowerCase() !== 'tidak aktif')
    .map(r => String(r[0]).trim());
}

// Simpan senarai jenis dokumen — tulis semula sheet sepenuhnya
function simpanJenis(senarai) {
  autoSetupDatabase();
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_JENIS);

  // Padam semua baris data lama
  if (sheet.getLastRow() > 1) {
    sheet.getRange(2, 1, sheet.getLastRow() - 1, 2).clearContent();
  }

  // Tulis semula
  if (senarai.length > 0) {
    const rows = senarai.map(nama => [nama, 'Aktif']);
    sheet.getRange(2, 1, rows.length, 2).setValues(rows);
  }
  return senarai;
}

}

function _simpanKonfigurasi(sheet, kunci, senarai) {
  const data = sheet.getDataRange().getValues();
  const nilai = senarai.join('||');
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === kunci) { sheet.getRange(i + 1, 2).setValue(nilai); return; }
  }
  sheet.appendRow([kunci, nilai]);
}

function _dapatkanKonfigurasi(sheet, kunci, defaults) {
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === kunci) {
      const val = String(data[i][1] || '');
      return val ? val.split('||').map(j => j.trim()).filter(j => j) : [];
    }
  }
  _simpanKonfigurasi(sheet, kunci, defaults);
  return defaults;
}

// PENGURUSAN PENGGUNA (ADMIN)
// ============================================================
function simpanPengguna(data) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();

  const idx = rows.findIndex((r, i) => i > 0 && r[0] === data.originalId);
  const existingRow  = idx > 0 ? rows[idx] : [];
  const statusKekal  = existingRow[9] || STATUS_AKAUN.AKTIF;
  // Jika kata laluan null/kosong semasa sunting → kekalkan kata laluan sedia ada
  // Jika pengguna baharu dan tiada kata laluan → set nilai lalai (sepatutnya tidak berlaku kerana frontend validasi)
  const passwordFinal = (data.password && String(data.password).trim())
    ? String(data.password).trim()
    : (existingRow[2] || 'user123');

  const rowData = [
    data.id,
    data.name,
    passwordFinal,
    data.role,
    Array.isArray(data.modules) ? data.modules.join(',') : data.modules,
    existingRow[5] || data.id,  // Emel — guna ID sebagai emel jika kosong
    data.tel      !== undefined ? (data.tel      || '') : (existingRow[6] || ''),  // No_Tel
    data.jawatan  !== undefined ? (data.jawatan  || '') : (existingRow[7] || ''),  // Jawatan
    new Date().toISOString(),
    statusKekal,                 // Status_Akaun
    data.unit     !== undefined ? (data.unit     || '') : (existingRow[10] || ''), // Unit_Wad
  ];

  if (idx > 0) {
    sheet.getRange(idx + 1, 1, 1, rowData.length).setValues([rowData]);
  } else {
    sheet.appendRow(rowData);
  }
  return dapatkanPengguna();
}

function padamPenggunaAkaun(id) {
  if (id === 'admin') return dapatkanPengguna();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) { sheet.deleteRow(i + 1); break; }
  }
  return dapatkanPengguna();
}

function padamSemuaRekod(password) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const configSheet = ss.getSheetByName(SHEET_NAME_CONFIG);
  const configData = configSheet.getDataRange().getValues();
  let adminPass = 'admin123';
  
  for (let i = 1; i < configData.length; i++) {
    if (configData[i][0] === 'admin_password') { adminPass = configData[i][1]; break; }
  }
  
  if (password !== adminPass) return { success: false, message: 'Kata laluan salah.' };
  
  const rekodSheet = ss.getSheetByName(SHEET_NAME_REKOD);
  if (rekodSheet.getLastRow() > 1) {
    rekodSheet.deleteRows(2, rekodSheet.getLastRow() - 1);
  }
  const logSheet = ss.getSheetByName(SHEET_NAME_LOG);
  if (logSheet.getLastRow() > 1) {
    logSheet.deleteRows(2, logSheet.getLastRow() - 1);
  }
  return { success: true };
}

const SHEET_NAME_DALAMAN = 'Serahan_Dalaman';
const SHEET_NAME_SD_CHAT = 'Serahan_Dalaman_Chat';
const HEADERS_DALAMAN = [
  'ID_Serahan','Daripada_ID','Daripada_Nama','Daripada_Peranan',
  'Kepada_ID','Kepada_Nama','Kepada_Peranan',
  'Perihal','Catatan','Status',
  'Tarikh_Hantar','Tarikh_Baca','Tarikh_Sahkan','Catatan_Sahkan','Tarikh_Kemaskini',
  'Jenis_Dokumen'  // kolum ke-16, ditambah di hujung untuk migrasi selamat
];
const HEADERS_SD_CHAT = ['ID_Serahan','User_ID','Nama','Peranan','Mesej','Tarikh_Masa'];

function _ensureSerahanDalamanSheet(ss) {
  _syncSheet(ss, SHEET_NAME_DALAMAN, HEADERS_DALAMAN, '#0f766e', '#ffffff', '-');
  _syncSheet(ss, SHEET_NAME_SD_CHAT, HEADERS_SD_CHAT, '#0f766e', '#ffffff', '-');
}

// Dapatkan semua serahan dalaman milik pengguna, lengkap dengan chat setiap satu
function dapatkanSerahanDalaman(userId, peranan) {
  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  _ensureSerahanDalamanSheet(ss);
  const sheet = ss.getSheetByName(SHEET_NAME_DALAMAN);
  if (!sheet || sheet.getLastRow() <= 1) return [];

  // Ambil semua chat sekali sahaja, kumpul ikut ID_Serahan
  const chatSheet = ss.getSheetByName(SHEET_NAME_SD_CHAT);
  const chatMap = {};
  if (chatSheet && chatSheet.getLastRow() > 1) {
    const chatData = chatSheet.getRange(2, 1, chatSheet.getLastRow() - 1, HEADERS_SD_CHAT.length).getValues();
    chatData.forEach(r => {
      if (!r[0]) return;
      if (!chatMap[r[0]]) chatMap[r[0]] = [];
      chatMap[r[0]].push({ userId: r[1], nama: r[2], peranan: r[3], teks: r[4], masa: r[5] });
    });
  }

  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, HEADERS_DALAMAN.length).getValues();
  return data.filter(r => r[0] !== '').map(r => ({
    id:              r[0],
    daripadaId:      r[1],
    daripadaNama:    r[2],
    daripadaPeranan: r[3],
    kepadaId:        r[4],
    kepadaNama:      r[5],
    kepadaPeranan:   r[6],
    perihal:         r[7],
    catatan:         r[8],
    status:          r[9],
    tarikhHantar:    r[10],
    tarikhBaca:      r[11],
    tarikhSahkan:    r[12],
    catatanSahkan:   r[13],
    tarikhKemaskini: r[14],
    jenisDokumen:    r[15] || '',
    chat:            chatMap[r[0]] || []
  }))
  .filter(r => r.daripadaId === userId || r.kepadaId === userId)
  .reverse();
}

function hantarSerahanDalaman(data) {
  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  _ensureSerahanDalamanSheet(ss);
  const sheet = ss.getSheetByName(SHEET_NAME_DALAMAN);

  const now = new Date().toLocaleString('ms-MY', {
    day:'2-digit', month:'short', year:'numeric', hour:'2-digit', minute:'2-digit'
  });
  const row = [
    data.id, data.daripadaId, data.daripadaNama, data.daripadaPeranan,
    data.kepadaId, data.kepadaNama, data.kepadaPeranan,
    data.perihal, data.catatan || '-', 'Baharu',
    now, '-', '-', '-', now,
    data.jenisDokumen || '-'
  ];
  sheet.appendRow(row);
  return dapatkanSerahanDalaman(data.daripadaId, data.daripadaPeranan);
}

function tandaBacaSerahanDalaman(sdId, userId, peranan) {
  autoSetupDatabase();
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  _ensureSerahanDalamanSheet(ss);
  const sheet = ss.getSheetByName(SHEET_NAME_DALAMAN);
  const data  = sheet.getDataRange().getValues();
  const now   = new Date().toLocaleString('ms-MY', {
    day:'2-digit', month:'short', year:'numeric', hour:'2-digit', minute:'2-digit'
  });
  for (let i = 1; i < data.length; i++) {

  autoSetupDatabase();

  _ensureSerahanDalamanSheet(ss);
  const sheet = ss.getSheetByName(SHEET_NAME_DALAMAN);
  const data  = sheet.getDataRange().getValues();
  const now   = new Date().toLocaleString('ms-MY', {day:'2-digit',month:'short',year:'numeric',hour:'2-digit',minute:'2-digit'});

// Semak sama ada emel wujud & akaun aktif — Langkah 1 "Lupa Kata Laluan"
function semakEmelWujud(emel) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();
  const emelBersih = String(emel || '').trim().toLowerCase();

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][5]).trim().toLowerCase() === emelBersih) {   // kolum 6 = Emel
      const status = String(rows[i][9] || STATUS_AKAUN.AKTIF).trim() || STATUS_AKAUN.AKTIF; // kolum 10 = Status_Akaun
      if (status === STATUS_AKAUN.MENUNGGU) {
        return { success: false, message: 'Akaun ini masih menunggu kelulusan Pentadbir Sistem.' };
      }
      if (status === STATUS_AKAUN.DITOLAK) {
        return { success: false, message: 'Akaun ini tidak aktif. Sila hubungi Pentadbir Sistem.' };
      }
      return { success: true, nama: rows[i][1] };
    }
  }
  return { success: false, message: 'Emel ini tidak ditemui dalam sistem.' };
}

// Tetapkan semula kata laluan terus melalui emel (tanpa kata laluan lama) — Langkah 2
function resetKataLaluanEmel(emel, passBaharu) {
  autoSetupDatabase();
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME_PENGGUNA);
  const rows  = sheet.getDataRange().getValues();
  const emelBersih = String(emel || '').trim().toLowerCase();

  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][5]).trim().toLowerCase() === emelBersih) {
      const status = String(rows[i][9] || STATUS_AKAUN.AKTIF).trim() || STATUS_AKAUN.AKTIF;
      if (status === STATUS_AKAUN.MENUNGGU || status === STATUS_AKAUN.DITOLAK) {
        return { success: false, message: 'Akaun ini tidak aktif. Sila hubungi Pentadbir Sistem.' };
      }
      sheet.getRange(i + 1, 3).setValue(passBaharu);                 // kolum 3 = Kata_Laluan
      sheet.getRange(i + 1, 9).setValue(new Date().toISOString());   // kolum 9 = Tarikh_Kemaskini
      return { success: true, nama: rows[i][1] };
    }
  }
  return { success: false, message: 'Emel ini tidak ditemui dalam sistem.' };
}
